#pragma pack(push, 1)
struct PacketHeader {
    uint16_t size;                 // 헤더 포함 전체 바이트 수
    uint16_t type;                 // 아래 번호 대역
};

// 타입 번호 대역 (제안):  1~49 로비 TCP / 50~99 인게임 UDP / 100~149 인게임 TCP
enum InGameTcpType : uint16_t {
    CS_GAME_EVENT = 100,           // 클라이언트 → 서버 : 내 이벤트
    SC_OPP_EVENT,                  // 서버 → 클라이언트 : 상대 이벤트 (본문은 CS_GAME_EVENT와 같음)
    SC_TIME_SYNC,
    SC_GAME_END,
    SC_OPP_LEFT,
};

enum EventType : uint8_t {
    BLOCK_HIT, BLOCK_BROKEN, ITEM_GET, ITEM_END,
    BALL_LOST, RESPAWN, STAGE_CLEAR, STAGE_START,
};

struct GameEventPacket : PacketHeader {   // CS_GAME_EVENT, SC_OPP_EVENT 공용 (20바이트)
    uint32_t timeMs;               // 게임 시작 후 경과 시간 (제한시간 판정 · 디버깅용)
    uint8_t  event;                // EventType
    uint8_t  arg8;                 // 아이템 종류, 스테이지 번호
    uint16_t arg16;                // 블록 인덱스, 남은 공 수
    int32_t  arg32;                // 남은 내구도, 얻은 점수, 대기 시간, 보너스 점수
    uint32_t score;                // 이 이벤트를 반영한 뒤의 내 총점
};

struct TimeSyncPacket : PacketHeader { uint16_t remainSec; };           // 6바이트
struct GameEndPacket  : PacketHeader {                                  // 14바이트
    uint8_t  result;               // 0 패배, 1 승리, 2 무승부 (받는 사람 기준)
    uint8_t  reason;               // 0 시간 종료, 1 모두 클리어, 2 상대 이탈
    uint32_t myScore;
    uint32_t oppScore;
};
#pragma pack(pop)

static_assert(sizeof(GameEventPacket) == 20, "패킷 크기 확인");
